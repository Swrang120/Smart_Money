// login.html me Reset/Forgot Password Submit Event handler ko isse replace karein: async function submitForgotPasswordModal() { const emailInput = document.getElementById("resetEmailInput"); // Aapke input ka ID const email = emailInput ? emailInput.value.trim().toLowerCase() : ""; if (!email) { alert("Please enter your registered email address."); return; } const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; if (!emailPattern.test(email)) { alert("Please enter a valid email address."); return; } try { const currentOrigin = window.location.origin; const redirectUrl = `${currentOrigin}/update-password.html`; // Supabase Auth Direct Mail Trigger (No EmailJS required) const { error } = await supabaseClient.auth.resetPasswordForEmail(email, { redirectTo: redirectUrl }); if (error) throw error; alert("✓ Password reset link has been sent to your email via Supabase! Please check your inbox and spam folder."); // Modal close karne ka code yahan rakhein (if applicable) closeResetModal(); } catch (err) { console.error("Supabase Reset Error:", err); alert("Error: " + (err.message || "Unable to send reset link. Please try again.")); } }
